# scripts/cfo_calc.py and scripts/test_cfo_calc.py need NOTHING from
# this file - they are standard library only, by design.
#
# openpyxl is required only to build or verify the Excel workbook:
#   scripts/build_dashboard.py   writes assets/cfo-premium-dashboard.xlsx
#   scripts/verify_signature.py  reads its document properties
#
# The version is pinned because openpyxl changes the archive layout
# between releases (comment part naming, shared strings), which changes
# the workbook's SHA-256 and breaks the SIGNATURE.json match even when
# the content is identical.
#
#   python3 -m pip install -r requirements.txt

# The pin matters for reproducibility: openpyxl changes archive layout between
# releases, so a different version produces a different SHA-256 for identical
# content and scripts/test_workbook_excel.py will report that a rebuild no
# longer matches the committed workbook.
#
# Verify any rebuild before shipping it:
#   python3 scripts/build_dashboard.py -o /tmp/rebuild.xlsx
#   CFO_WORKBOOK=/tmp/rebuild.xlsx python3 scripts/test_workbook_excel.py
openpyxl==3.1.5
